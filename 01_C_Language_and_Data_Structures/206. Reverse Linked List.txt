/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     struct ListNode *next;
 * };
 */
struct ListNode* reverseList(struct ListNode* head) {
    struct ListNode* prev = NULL;
    struct ListNode* cur = head;
    struct ListNode* last = NULL;


    while( cur != NULL){
        last = cur->next;
        cur->next = prev;

        prev = cur;
        cur = last;
    }
    return prev;
}
