bool isPalindrome(char* s) {
    int len = 0;
    while(s[len] != '\0'){
        len += 1;
    }

    int left = 0;
    int right = len-1;

    while(left < right){

        while(left < right && !isalnum(s[left])){
            left += 1;
        }
        while(left < right && !isalnum(s[right])){
            right -= 1;
        }
        if(tolower(s[left]) != tolower(s[right])){
            return false;
        }
        else{
            left += 1;
            right -= 1;
        }
    }
    return true;
}
